def _is_oob(query: str) -> bool:
    text = query.lower()
    return any(re.search(p, text, re.IGNORECASE) for p in _OOB_PATTERNS)


def _safe_int(val: Any) -> int | None:
    if val is None:
        return None
    try:
        return int(val)
    except (TypeError, ValueError):
        return None


def _clean_result(raw: dict[str, Any]) -> dict[str, Any]:
    """Normalize and validate the LLM JSON output."""
    intent = str(raw.get("intent") or "UNKNOWN").upper().strip()
    if intent not in VALID_INTENTS:
        intent = "UNKNOWN"

    in_scope = bool(raw.get("in_scope", True))
    deflection_reason = raw.get("deflection_reason") if not in_scope else None

    raw_params = raw.get("parameters") or {}
    if not isinstance(raw_params, dict):
        raw_params = {}

    group_by_raw = raw_params.get("group_by") or []
    group_by: list[str] = []
    if isinstance(group_by_raw, list):
        group_by = [str(g).lower() for g in group_by_raw if isinstance(g, str)]

    order_raw = str(raw_params.get("order") or "").upper().strip()
    order = order_raw if order_raw in ("ASC", "DESC") else None

    raw_filters = raw.get("filters") or {}
    if not isinstance(raw_filters, dict):
        raw_filters = {}

    opt_mode = str(raw_filters.get("optimization_mode") or "").upper().strip()
    if opt_mode not in ("SERVICE_LEVEL", "MARGIN", "SELL_THROUGH"):
        opt_mode = None  # type: ignore[assignment]

    decision = str(raw_filters.get("final_decision") or "").upper().strip()
    if decision not in ("REPLENISH", "MONITOR", "POLICY_REVIEW"):
        decision = None  # type: ignore[assignment]

    risk_level = str(raw_filters.get("risk_level") or "").upper().strip()
    if risk_level not in ("HIGH", "MEDIUM", "LOW"):
        risk_level = None  # type: ignore[assignment]

    months_back = _safe_int(raw_filters.get("months_back"))
    if months_back is not None and months_back <= 0:
        months_back = None

    # year_list: exactly two distinct valid 4-digit years for "X vs Y" comparisons
    year_list_raw = raw_filters.get("year_list")
    year_list: list[int] | None = None
    if isinstance(year_list_raw, list) and len(year_list_raw) == 2:
        parsed_years = [_safe_int(y) for y in year_list_raw]
        if all(isinstance(y, int) and 2000 <= y <= 2100 for y in parsed_years) and parsed_years[0] != parsed_years[1]:
            year_list = parsed_years

    _VALID_REGIONS = {"North", "South", "East", "West"}
    _VALID_CATEGORIES = {"Electronics", "Clothing", "Toys", "Furniture", "Groceries"}

    region_raw = raw_filters.get("region")
    region = str(region_raw).capitalize() if region_raw else None
    if region not in _VALID_REGIONS:
        region = None  # type: ignore[assignment]

    category_raw = raw_filters.get("category")
    category = str(category_raw).capitalize() if category_raw else None
    if category not in _VALID_CATEGORIES:
        category = None  # type: ignore[assignment]

    product_raw = raw_filters.get("product")
    product = str(product_raw).strip() if product_raw else None

    store_raw = raw_filters.get("store")
    store = str(store_raw).strip() if store_raw else None

    return {
        "intent": intent,
        "in_scope": in_scope,
        "deflection_reason": deflection_reason,
        "parameters": {
            "demand_increase_pct": _safe_int(raw_params.get("demand_increase_pct")),
            "lead_time_increase_days": _safe_int(raw_params.get("lead_time_increase_days")),
            "inventory_adjustment_pct": _safe_int(raw_params.get("inventory_adjustment_pct")),
            "limit": _safe_int(raw_params.get("limit")),
            "order": order,
            "group_by": group_by,
            "metric": (
                str(raw_params.get("metric")).lower().strip()
                if raw_params.get("metric")
                and str(raw_params.get("metric")).lower().strip() in _KNOWN_METRICS
                else None
            ),
        },
        "filters": {
            "category": category,
            "region": region,
            "year": _safe_int(raw_filters.get("year")),
            "year_list": year_list,
            "months_back": months_back,
            "optimization_mode": opt_mode,
            "final_decision": decision,
            "risk_level": risk_level,
            "product": product,
            "store": store,
        },
    }


def extract_intent_llm(query: str, conv_state: dict[str, Any] | None = None) -> dict[str, Any]:
    """Call LLM to classify intent and extract parameters. Falls back to UNKNOWN on any error."""
    if not isinstance(query, str) or not query.strip():
        return dict(_FALLBACK_RESULT)

    # Hard-coded OOB guard (applies even if LLM is unavailable)
    if _is_oob(query):
        return {
            **_FALLBACK_RESULT,
            "intent": "OUT_OF_SCOPE",
            "in_scope": False,
            "deflection_reason": "Query is outside ISANA's inventory risk domain",
        }

    # Build session context for the LLM.
    # General filters are excluded to prevent prior-turn regions/categories bleeding into
    # unrelated follow-on queries. Exception: when last_intent=SCENARIO_ANALYSIS, inject
    # prior scenario params + filters so continuation queries like "same scenario in
    # SERVICE_LEVEL" can inherit them without the user re-stating values.
    ctx_summary = "none"
    if isinstance(conv_state, dict):
        last_intent = conv_state.get("active_intent") or conv_state.get("last_intent")
        if last_intent:
            ctx_summary = f"last_intent={last_intent}"
            _last_upper = str(last_intent).upper()
            if _last_upper == "SCENARIO_ANALYSIS":
                last_resolved = conv_state.get("last_resolved") or {}
                last_params = last_resolved.get("parameters") or {}
                last_filters = last_resolved.get("filters") or {}
                prior_scenario: dict[str, Any] = {
                    k: last_params[k]
                    for k in ("demand_increase_pct", "lead_time_increase_days", "inventory_adjustment_pct")
                    if last_params.get(k) is not None
                }
                if prior_scenario:
                    ctx_summary += f"; prior_scenario_params={prior_scenario}"
                for key in ("category", "region", "optimization_mode"):
                    if last_filters.get(key):
                        ctx_summary += f"; prior_{key}={last_filters[key]}"
            elif _last_upper in ("ANALYTICAL_QUERY", "RISK_QUERY"):
                last_resolved = conv_state.get("last_resolved") or {}
                last_params = last_resolved.get("parameters") or {}
                prior_metric = last_params.get("metric")
                prior_group_by = last_params.get("group_by") or []
                if prior_metric:
                    ctx_summary += f"; prior_metric={prior_metric}"
                if prior_group_by:
                    ctx_summary += f"; prior_group_by={prior_group_by}"
                # NOTE: spatial filters (region, category) are intentionally NOT injected here.
                # Carrying prior_region/prior_category causes session bleed — fresh analytical
                # queries should not inherit spatial scope from a previous turn. Continuation
                # is signalled explicitly by the user with tokens like "now", "same", "also".

    prompt = _USER_PROMPT_TEMPLATE.format(query=query.strip(), context=ctx_summary)

    try:
        from agent_layer.llm_gateway import _execute_chat
        raw_text = _execute_chat(prompt, system_prompt=_SYSTEM_PROMPT)

        # Strip markdown fences if present
        raw_text = re.sub(r"^```(?:json)?\s*", "", raw_text.strip(), flags=re.IGNORECASE)
        raw_text = re.sub(r"\s*```$", "", raw_text.strip())

        raw_json = json.loads(raw_text)
        result = _clean_result(raw_json)

        # Secondary OOB guard: if LLM returned in_scope=False, trust it
        return result

    except Exception:
        # LLM unavailable or parse error — return fallback, let regex interpreter handle it
        return dict(_FALLBACK_RESULT)
