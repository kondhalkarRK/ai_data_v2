  B) Identify numeric columns (measures)
  C) Identify text columns (dimensions)
  D) Detect result type:
     - 1 row, 1 col → scalar
     - 1 dimension, 1 measure → ranked list
     - 2+ dimensions → grouped
     - time column present → trend
     - 2 measures → comparison
  
  E) Generate appropriate text:
  
  SCALAR:
  headline = f"Total {metric_label}"
  narrative = f"The total {metric_label} 
    across all records is {value_formatted}."
  
  RANKED LIST (most common):
  top = df.iloc[0]
  bottom = df.iloc[-1]
  total = df[value_col].sum()
  top_pct = top[value_col] / total * 100
  
  headline = f"{metric_label} by {dim_label}"
  narrative = (
    f"Across {len(df)} {dim_label} values, "
    f"{top[dim_col]} leads with "
    f"{format_value(top[value_col])} "
    f"({top_pct:.0f}% of total). "
    f"{bottom[dim_col]} is lowest at "
    f"{format_value(bottom[value_col])}. "
    f"Total {metric_label}: "
    f"{format_value(total)}."
  )
  
  key_findings = [
    f"{top[dim_col]} holds the top position 
      with {top_pct:.0f}% share",
    f"Top 3 account for 
      {top3_pct:.0f}% of total",
    f"Range spans from 
      {format_value(bottom[value_col])} 
      to {format_value(top[value_col])}"
  ]
  
  TREND:
  latest = df.iloc[-1]
  prior = df.iloc[-2] if len(df)>1 else None
  
  if prior:
    change = (latest[val]-prior[val])/prior[val]*100
    direction = "up" if change > 0 else "down"
    
  narrative = (
    f"The {metric_label} trend shows "
    f"{len(df)} periods. "
    f"The most recent period "
    f"({latest[time_col]}) reached "
    f"{format_value(latest[val_col])}"
    f"{f', {direction} {abs(change):.1f}% "
       f"from prior period' if prior else ''}. "
    f"Average across all periods: "
    f"{format_value(df[val_col].mean())}."
  )
  
  GROUPED (2 dimensions):
  narrative = (
    f"Breaking down {metric_label} across "
    f"{len(df)} combinations of "
    f"{dim1_label} and {dim2_label}. "
    f"{top[dim1_col]} / {top[dim2_col]} "
    f"leads at "
    f"{format_value(top[val_col])}."
  )

METHOD format_value(value, format_type) → str:
  currency: £1,234,567 or £1.2M for large
  integer: 1,234
  percent: 12.3%
  decimal: 1,234.56
  
  For currency over 1M:
    use £1.2M format
  For currency over 1K:
    use £42.3K format
  For currency under 1K:
    use £423 format

METHOD should_auto_narrate(question) → bool:
  Returns True for:
  why, explain, describe, tell me about,
  what does this mean, analyse, analysis,
  insights, summary, overview, understand,
  what if (always narrate what-if)
